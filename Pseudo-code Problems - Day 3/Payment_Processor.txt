Class Transaction<T>
    Attribute: transactionId
               T paymentMethod
    Constructor(transactionId, paymentMethod)

    Method process()
        print "Processing transaction " + transactionId 
              + " using " + paymentMethod

Class UPI
    Attribute: upiId
    toString() → return "UPI: " + upiId

Class Card
    Attribute: cardNumber
    toString() → return "Card: ****" + last4 digits

Class NetBanking
    Attribute: bankName
    toString() → return "NetBanking: " + bankName

Main
    create Transaction<UPI>("T1", new UPI("user@upi"))
    create Transaction<Card>("T2", new Card("1234567890123456"))
    create Transaction<NetBanking>("T3", new NetBanking("SBI"))

    call process() for each transaction
