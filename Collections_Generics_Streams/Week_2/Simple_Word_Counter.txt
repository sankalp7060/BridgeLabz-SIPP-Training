function wordCounter(String text) {
    String[] words=text.toLowerCase().replaceAll("[^a-z ]","").split(" ")
    Map<String,Integer> freq=new HashMap<>()
    for w in words {
        freq.put(w,freq.getOrDefault(w,0)+1)
    }
    List<Map.Entry<String,Integer>> sorted=freq.entrySet().stream()
        .sorted((a,b)->b.value-a.value)
        .toList()
    for e in sorted print(e.key+" "+e.value)
}

function main() {
    wordCounter("The news today is about the rise in technology trends.")
}
